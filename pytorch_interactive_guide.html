🐍 مقدمة عن PyTorch

PyTorch هي مكتبة مفتوحة المصدر للتعلم العميق طوّرتها Meta (Facebook AI Research). تُستخدم على نطاق واسع في البحث الأكاديمي والصناعي لبناء وتدريب الشبكات العصبية.

🧮 موترات (Tensors) 🔄 تفاضل تلقائي 🧠 شبكات عصبية ⚡ دعم GPU 🌐 معالجة موزعة 📱 نشر على الموبايل

لماذا PyTorch؟

  • سهولة التعلم: واجهة Pythonic بديهية
  • رسم بياني ديناميكي: تعريف النموذج وتنفيذه في آنٍ واحد
  • مجتمع ضخم: آلاف النماذج الجاهزة والمكتبات الإضافية
  • بحث وإنتاج: من النموذج الأولي إلى النشر

🧮 الموترات (Tensors)

الموتر (Tensor) هو البنية الأساسية في PyTorch — مصفوفة متعددة الأبعاد تشبه numpy.ndarray لكن مع دعم GPU وتتبع التدرجات.

🎯 إنشاء الموترات

Python
import torch # === من بيانات مباشرة === a = torch.tensor([1, 2, 3, 4, 5]) b = torch.tensor([[1., 2.], [3., 4.]]) print("من قائمة:", a) print("ثنائي الأبعاد:\n", b) # === مصفوفات مهيأة === print("\nzeros(3,4):\n", torch.zeros(3, 4)) print("ones(2,3):\n", torch.ones(2, 3)) print("full(2,2, 7):\n", torch.full((2, 2), 7.0)) print("eye(3):\n", torch.eye(3)) # === سلاسل عددية === print("\narange(0, 10, 2):", torch.arange(0, 10, 2)) print("linspace(0, 1, 5):", torch.linspace(0, 1, 5)) # === عشوائي === torch.manual_seed(42) print("\nrand(2,3):\n", torch.rand(2, 3)) print("randn(2,3):\n", torch.randn(2, 3)) print("randint(0,10,(2,3)):\n", torch.randint(0, 10, (2, 3))) # === أنواع البيانات === print("\nfloat32:", torch.tensor([1,2,3], dtype=torch.float32).dtype) print("int64:", torch.tensor([1,2,3], dtype=torch.int64).dtype) print("bool:", torch.tensor([1,0,1], dtype=torch.bool).dtype)

🔢 العمليات الحسابية

Python
import torch x = torch.tensor([[1., 2.], [3., 4.]]) y = torch.tensor([[5., 6.], [7., 8.]]) print("X:\n", x, "\nY:\n", y) print("\n--- عمليات أساسية ---") print("X+Y:\n", x + y) print("X-Y:\n", x - y) print("X*Y:\n", x * y) print("X@Y (ضرب مصفوفات):\n", x @ y) print("\n--- دوال رياضية ---") print("X**2:\n", x.pow(2)) print("sqrt(X):\n", torch.sqrt(x)) print("exp(X):\n", torch.exp(x)) print("\n--- اختزال ---") print("sum:", x.sum(), "| mean:", x.mean()) print("max:", x.max(), "| min:", x.min()) print("\n--- تغيير الشكل ---") z = torch.arange(12) print("reshape(3,4):\n", z.reshape(3, 4)) print("view(4,3):\n", z.view(4, 3)) print("\n--- دمج ---") a, b = torch.tensor([1,2,3]), torch.tensor([4,5,6]) print("cat:", torch.cat([a, b])) print("stack dim=0:\n", torch.stack([a, b], dim=0))

🔄 التفاضل التلقائي (Autograd)

أهم ميزة في PyTorch! نظام autograd يسجل كل العمليات تلقائياً ويحسب التدرجات.

Python
import torch # === أساسيات autograd === x = torch.tensor([1., 2., 3.], requires_grad=True) print("x:", x) y = x ** 2 z = y.sum() print(f"y = x² = {y}") print(f"z = sum(y) = {z.item()}") z.backward() print(f"🔑 dz/dx = {x.grad}") # [2, 4, 6] # === انحدار خطي بسيط === print("\n" + "="*50) print("📈 انحدار خطي بسيط") X = torch.tensor([[1.], [2.], [3.], [4.]], dtype=torch.float32) Y = torch.tensor([[2.], [4.], [6.], [8.]], dtype=torch.float32) w = torch.tensor([[0.]], requires_grad=True) b = torch.tensor([[0.]], requires_grad=True) lr = 0.01 for epoch in range(100): y_pred = X @ w + b loss = ((y_pred - Y) ** 2).mean() loss.backward() with torch.no_grad(): w -= lr * w.grad b -= lr * b.grad w.grad.zero_() b.grad.zero_() if epoch % 20 == 0: print(f"Epoch {epoch:3d}: loss={loss.item():.6f}, w={w.item():.4f}, b={b.item():.4f}") print(f"\n✅ النتيجة: w={w.item():.4f}, b={b.item():.4f} (الهدف: Y=2X)")

💡 فهم autograd

ماذا يحدث عندما نستدعي .backward()؟

  1. ينتقل عكسياً عبر الرسم البياني الحسابي
  2. يحسب مشتقة كل عقدة باستخدام قاعدة السلسلة
  3. يُخزّن النتيجة في .grad

⚠️ التدرجات تتراكم! استخدم zero_grad() قبل كل خطوة.

🧠 nn.Module - أساس الشبكات العصبية

كل شبكة عصبية في PyTorch ترث من nn.Module. هذه الفئة تدير المعاملات والطبقات تلقائياً.

Python
import torch import torch.nn as nn # تعريف شبكة عصبية بسيطة class SimpleNN(nn.Module): def __init__(self, input_size, hidden_size, output_size): super().__init__() self.fc1 = nn.Linear(input_size, hidden_size) self.relu = nn.ReLU() self.fc2 = nn.Linear(hidden_size, output_size) self.dropout = nn.Dropout(0.2) def forward(self, x): x = self.fc1(x) x = self.relu(x) x = self.dropout(x) x = self.fc2(x) return x model = SimpleNN(784, 256, 10) print("📐 هيكل النموذج:") print(model) print(f"\n🔢 عدد المعاملات: {sum(p.numel() for p in model.parameters()):,}") x = torch.randn(4, 784) output = model(x) print(f"\n📥 إدخال: {x.shape}") print(f"📤 إخراج: {output.shape}") # Sequential - طريقة مختصرة model_seq = nn.Sequential( nn.Linear(784, 256), nn.ReLU(), nn.Dropout(0.2), nn.Linear(256, 10) ) print(f"\n🔗 Sequential output: {model_seq(x).shape}") model.train() print(f"🔄 وضع التدريب: {model.training}") model.eval() print(f"🔄 وضع التقييم: {model.training}")

🔲 طبقات الالتفاف (Convolution Layers)

Python
import torch import torch.nn as nn # صورة: batch=2, channels=3 (RGB), 32x32 input_image = torch.randn(2, 3, 32, 32) print(f"📥 إدخال: {input_image.shape}") conv = nn.Conv2d(in_channels=3, out_channels=16, kernel_size=3, stride=1, padding=1) output = conv(input_image) print(f"📤 Conv2d(3→16, k=3, p=1): {output.shape}") conv2 = nn.Conv2d(16, 32, kernel_size=3, stride=2, padding=1) output2 = conv2(output) print(f"📤 Conv2d(16→32, k=3, s=2): {output2.shape}") print("\n📐 قاعدة حساب أبعاد الإخراج:") print(" H_out = (H_in + 2×P - K) / S + 1") print(f" (32+2-3)/1+1 = {(32+2-3)//1+1}") print(f" (32+2-3)/2+1 = {(32+2-3)//2+1}") # ConvTranspose2d conv_trans = nn.ConvTranspose2d(32, 16, kernel_size=4, stride=2, padding=1) output3 = conv_trans(output2) print(f"\n🔄 ConvTranspose2d: {output2.shape} → {output3.shape}") print(f"\n🔢 معاملات أول طبقة: {conv.weight.numel()}") print(f" = 16×3×3×3 = {16*3*3*3}")

Python
import torch import torch.nn as nn x = torch.linspace(-3, 3, 13) print("الإدخال x:", x.tolist()) activations = { 'ReLU': nn.ReLU(), 'LeakyReLU': nn.LeakyReLU(0.1), 'Sigmoid': nn.Sigmoid(), 'Tanh': nn.Tanh(), 'GELU': nn.GELU(), 'SiLU': nn.SiLU(), 'Softplus': nn.Softplus(), 'ELU': nn.ELU(), } print("\n📊 مقارنة دوال التنشيط:") print(f"{'الدالة':<12} {'f(-3)':>8} {'f(-1)':>8} {'f(0)':>8} {'f(1)':>8} {'f(3)':>8}") print("-" * 52) for name, fn in activations.items(): y = fn(x) vals = [y[i].item() for i in [0, 4, 6, 8, 12]] print(f"{name:<12} {vals[0]:8.3f} {vals[1]:8.3f} {vals[2]:8.3f} {vals[3]:8.3f} {vals[4]:8.3f}") # Softmax print("\n📊 Softmax:") logits = torch.tensor([2.0, 1.0, 0.1, -1.0]) sm = nn.Softmax(dim=0)(logits) print(f"المدخلات: {logits.tolist()}") print(f"المخرجات: {[f'{v:.4f}' for v in sm.tolist()]}") print(f"المجموع = {sm.sum().item():.6f}")

⚡ المُحسِّنات (Optimizers)

Python
import torch import torch.nn as nn import torch.optim as optim # مقارنة SGD vs Adam على f(x) = (x-3)² criterion = nn.MSELoss() target = torch.tensor([3.0]) def train_opt(opt_class, name, **kw): x = torch.tensor([0.0], requires_grad=True) opt = opt_class([x], **kw) hist = [] for _ in range(50): loss = criterion(x, target) opt.zero_grad() loss.backward() opt.step() hist.append(x.item()) print(f"{name:<14}: x → {x.item():.6f} | أول 5: {[f'{v:.2f}' for v in hist[:5]]}") train_opt(optim.SGD, 'SGD', lr=0.1) train_opt(optim.SGD, 'SGD+momentum', lr=0.1, momentum=0.9) train_opt(optim.Adam, 'Adam', lr=0.1) train_opt(optim.AdamW, 'AdamW', lr=0.1) train_opt(optim.RMSprop, 'RMSprop', lr=0.1) # جدولة معدل التعلم print("\n📅 جدولة StepLR(step=10, gamma=0.5):") model = nn.Linear(10, 1) opt = optim.SGD(model.parameters(), lr=0.1) sch = optim.lr_scheduler.StepLR(opt, step_size=10, gamma=0.5) lrs = [] for _ in range(30): lrs.append(opt.param_groups[0]['lr']) sch.step() print(f" {[f'{lr:.3f}' for lr in lrs[::5]]}") print("\n📊 دليل الاختيار:") print(" Adam/AdamW: الخيار الافتراضي الأفضل") print(" SGD+momentum: دقة أعلى أحياناً") print(" RMSprop: مناسب لـ RNNs")

⚡ torch.compile - تسريع النموذج

أسهل طريقة لتسريع نموذج PyTorch (حتى 2x أسرع!) بسطر واحد.

Python
import torch import torch.nn as nn model = nn.Sequential( nn.Linear(512, 256), nn.GELU(), nn.Linear(256, 128), nn.GELU(), nn.Linear(128, 10) ) try: compiled_model = torch.compile(model) print("✅ تم تجميع النموذج بنجاح!") x = torch.randn(32, 512) with torch.no_grad(): out1 = model(x) out2 = compiled_model(x) print(f"📊 الفرق الأقصى: {(out1 - out2).abs().max().item():.10f}") print(" ✅ النتائج متطابقة!") except Exception as e: print(f"⚠️ torch.compile غير متاح: محاكاة") print(""" 📋 خيارات torch.compile: mode='default' - توازن السرعة والتجميع mode='reduce-overhead' - أسرع للدفعات الصغيرة mode='max-autotune' - أسرع أداء fullgraph=True - رسم بياني كامل dynamic=False - أشكال ثابتة (أسرع) """)

import torch import torch.nn as nn class CIFAR10_CNN(nn.Module): def __init__(self, num_classes=10): super().__init__() self.block1 = nn.Sequential(nn.Conv2d(3,32,3,padding=1),nn.BatchNorm2d(32),nn.ReLU(True),nn.MaxPool2d(2)) self.block2 = nn.Sequential(nn.Conv2d(32,64,3,padding=1),nn.BatchNorm2d(64),nn.ReLU(True),nn.MaxPool2d(2)) self.block3 = nn.Sequential(nn.Conv2d(64,128,3,padding=1),nn.BatchNorm2d(128),nn.ReLU(True),nn.MaxPool2d(2)) self.classifier = nn.Sequential(nn.AdaptiveAvgPool2d(1),nn.Flatten(),nn.Dropout(0.3),nn.Linear(128,64),nn.ReLU(True),nn.Dropout(0.2),nn.Linear(64,num_classes)) def forward(self, x): return self.classifier(self.block3(self.block2(self.block1(x)))) model = CIFAR10_CNN(10) x = torch.randn(4, 3, 32, 32) output = model(x) print("📐 هيكل النموذج:") print(model) print(f"\n✅ إدخال: {x.shape} → إخراج: {output.shape}") print(f"🔢 عدد المعاملات: {sum(p.numel() for p in model.parameters()):,}") probs = torch.softmax(output, dim=1) print(f"🏷️ التوقعات: {probs.argmax(dim=1).tolist()}")

  3. تدريب 30 حقبة مع تتبع الخسارة والدقة

import torch, torch.nn as nn, torch.optim as optim from torch.utils.data import DataLoader, TensorDataset torch.manual_seed(42) n_samples = 1000 X = torch.randn(n_samples, 20) y = torch.randint(0, 3, (n_samples,)) n_train = int(0.8 * n_samples) train_ds = TensorDataset(X[:n_train], y[:n_train]) val_ds = TensorDataset(X[n_train:], y[n_train:]) train_loader = DataLoader(train_ds, batch_size=32, shuffle=True) val_loader = DataLoader(val_ds, batch_size=32) print(f"📊 تدريب={len(train_ds)}, تحقق={len(val_ds)}") model = nn.Sequential(nn.Linear(20,64),nn.ReLU(),nn.Dropout(0.3),nn.Linear(64,32),nn.ReLU(),nn.Dropout(0.2),nn.Linear(32,3)) criterion = nn.CrossEntropyLoss() optimizer = optim.Adam(model.parameters(), lr=0.01) scheduler = optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=30) train_losses, val_accs = [], [] for epoch in range(30): model.train() epoch_loss = 0 for bx, by in train_loader: optimizer.zero_grad() loss = criterion(model(bx), by) loss.backward() optimizer.step() epoch_loss += loss.item() scheduler.step() train_losses.append(epoch_loss/len(train_loader)) model.eval() correct = total = 0 with torch.no_grad(): for bx, by in val_loader: preds = model(bx).argmax(1) total += by.size(0) correct += (preds == by).sum().item() val_accs.append(correct/total) if epoch % 5 == 0: print(f"Epoch {epoch:2d}: loss={train_losses[-1]:.4f}, val_acc={val_accs[-1]:.3f}") print(f"\n✅ أفضل دقة: {max(val_accs):.3f} | نهائية: {val_accs[-1]:.3f}") print(f"📈 loss: {train_losses[0]:.4f} → {train_losses[-1]:.4f}") print("📊 رسم بياني (accuracy per epoch):") for i, acc in enumerate(val_accs): print(f" Epoch {i:2d}: {'█'*int(acc*50)} {acc:.3f}")

✏️ تمرين ٣: نقل التعلم (Transfer Learning)

🎯 المطلوب

  1. حمّل ResNet-18 مُدرَّب مسبقاً
  2. جمّد كل الطبقات ما عدا الأخيرة
  3. استبدل المُصنِّف بـ 5 فئات
استخدم param.requires_grad = False للتجميد. استبدل model.fc بطبقة Linear جديدة.
Python - الحل
import torch, torch.nn as nn class SimResNet18(nn.Module): def __init__(self, nc=1000): super().__init__() self.conv1 = nn.Conv2d(3,64,7,2,3); self.bn1 = nn.BatchNorm2d(64); self.relu = nn.ReLU(); self.maxpool = nn.MaxPool2d(3,2,1) self.layer1 = nn.Sequential(nn.Conv2d(64,64,3,1,1),nn.BatchNorm2d(64),nn.ReLU()) self.layer2 = nn.Sequential(nn.Conv2d(64,128,3,2,1),nn.BatchNorm2d(128),nn.ReLU()) self.avgpool = nn.AdaptiveAvgPool2d(1); self.fc = nn.Linear(128, nc) def forward(self,x): x = self.maxpool(self.relu(self.bn1(self.conv1(x)))) x = self.layer2(self.layer1(x)); x = self.avgpool(x) return self.fc(torch.flatten(x,1)) print("📥 تحميل ResNet-18 المُدرَّب مسبقاً...") model = SimResNet18(1000) total = sum(p.numel() for p in model.parameters()) print(f"🔢 إجمالي المعاملات: {total:,}") print("\n🔒 تجميد كل الطبقات...") frozen = 0 for p in model.parameters(): p.requires_grad = False frozen += p.numel() print(f" مجمّدة: {frozen:,}") print("\n🔄 استبدال المُصنِّف...") model.fc = nn.Sequential(nn.Dropout(0.4),nn.Linear(128,256),nn.ReLU(),nn.Dropout(0.3),nn.Linear(256,5)) trainable = sum(p.numel() for p in model.parameters() if p.requires_grad) print(f"📊 قابلة للتدريب: {trainable:,} ({100*trainable/total:.1f}%)") x = torch.randn(2, 3, 224, 224) out = model(x) print(f"\n✅ إدخال: {x.shape} → إخراج: {out.shape}") optimizer = torch.optim.Adam(filter(lambda p: p.requires_grad, model.parameters()), lr=0.001) n_opt = sum(p.numel() for g in optimizer.param_groups for p in g['params']) print(f"⚡ مُحسِّن: Adam على {n_opt:,} مُعامِل")

✏️ تمرين ٤: RNN لتحليل المشاعر

🎯 المطلوب

  1. طبقة Embedding لتحويل الكلمات
  2. طبقتان LSTM ثنائية الاتجاه
  3. Linear للتصنيف الثنائي (إيجابي/سلبي)
nn.Embedding(vocab, embed) + nn.LSTM(embed, hidden, num_layers=2, batch_first=True, bidirectional=True)
Python - الحل
import torch, torch.nn as nn class SentimentLSTM(nn.Module): def __init__(self, vocab=10000, embed=256, hidden=128, layers=2, classes=2, drop=0.3): super().__init__() self.embedding = nn.Embedding(vocab, embed, padding_idx=0) self.lstm = nn.LSTM(embed, hidden, layers, batch_first=True, bidirectional=True, dropout=drop if layers>1 else 0) self.cls = nn.Sequential(nn.Linear(hidden*2,64),nn.ReLU(),nn.Dropout(drop),nn.Linear(64,classes)) def forward(self, x): emb = self.embedding(x) _, (hidden, _) = self.lstm(emb) last = torch.cat((hidden[-2], hidden[-1]), dim=1) return self.cls(last) model = SentimentLSTM() x = torch.randint(1, 10000, (16, 50)) x[0, 30:] = 0 out = model(x) probs = torch.softmax(out, dim=1) print(f"📐 LSTM لتحليل المشاعر") print(f" إدخال: {x.shape} (batch=16, seq=50)") print(f" إخراج: {out.shape}") print(f"\n🏷️ التوقعات (أول 5):") for i in range(5): n, p = probs[i].tolist() s = "إيجابي 😊" if p > n else "سلبي 😞" print(f" {i+1}: {s} (إيجابي={p:.3f}, سلبي={n:.3f})") total_p = sum(p.numel() for p in model.parameters()) print(f"\n🔢 المعاملات: {total_p:,}") print(f" Embedding: {model.embedding.weight.numel():,}") print(f" LSTM: {sum(p.numel() for p in model.lstm.parameters()):,}") print(f" Classifier: {sum(p.numel() for p in model.cls.parameters()):,}")
